    instruments ¶

    Instrument management commands — mzt instruments list|check.

    These commands let users discover available instruments, check their readiness, and diagnose configuration issues. They build on the InstrumentProfileLoader and InstrumentRegistry — YAML profiles loaded from built-in, organization, and venue directories.

    The music metaphor: these commands are the instrument workshop's front desk. list shows what's on the rack. check picks one up and plays a test note.

    @classes_app.command(name="check")
def check_classes(
    class_name: str | None = typer.Option(
        None, "--class", help="Require this class to be runnable."
    ),
) -> None:
    """Refuse broken layers and configured chains with no available entry."""
    report, broken = _class_report()
    if class_name is not None:
        selected = report["classes"].get(class_name)
        if selected is None or not any(item["availability"] for item in selected["chain"]):

        raise typer.Exit(1)
    console.print(f"Checked {len(report['classes'])} capability classes")

    @classes_app.command(name="tiers")
def show_tier_classes(
    catalog_dir: Path | None = typer.Option(
        None, "--catalog", help="Catalog v2 directory (default: the checkout's plugins submodule)."
    ),
    write_proposal: Path | None = typer.Option(
        None,
        "--write-proposal",
        help="Write a classes-file fragment you can merge by hand. Never the user layer.",
    ),
    force: bool = typer.Option(False, "--force", help="Replace an existing proposal file."),
    json_output: bool = typer.Option(False, "--json"),
) -> None:
    """Print the frontier/balanced/light chains this machine resolves from the catalog.

    Read-only: this never writes ``~/.marianne/classes.yaml`` (your ordering is
    yours). ``--write-proposal PATH`` writes a fragment for you to merge.
    """
    from marianne.core.config.catalog import CatalogError, load_catalog
    from marianne.instruments.classes import class_source_paths
    from marianne.instruments.tier_classes import (
        TIER_CLASSES,
        render_proposal,
        tier_chains,
    )

    directory = catalog_dir or _default_catalog_dir()
    if directory is None:
        output_error("No catalog v2 found; pass --catalog DIR (plugins/marianne/docs/ref/catalog)")
        raise typer.Exit(1)
    try:
        catalog = load_catalog(directory)
    except CatalogError as exc:
        output_error(str(exc))
        raise typer.Exit(1) from exc

    profiles = _load_all_profiles()
    registry = InstrumentRegistry()
    for profile in profiles.values():
        registry.register(profile)
    chains = tier_chains(catalog, profiles_available=profiles)

    if write_proposal is not None:
        layers = class_source_paths()
        if write_proposal.expanduser().resolve() in {layers[1], layers[2]}:
            output_error(
                f"{write_proposal} is a class layer file; the proposal is for you to merge, "
                "so write it elsewhere"
            )
            raise typer.Exit(1)
        if write_proposal.exists() and not force:
            output_error(f"{write_proposal} exists; use --force to replace it")
            raise typer.Exit(1)
        note = (
            f"catalog {directory} (generated_at {catalog.index.generated_at}); "
            f"{len(profiles)} registered profiles; non-provisional tiers, no metered routes."
        )
        write_proposal.parent.mkdir(parents=True, exist_ok=True)
        write_proposal.write_text(render_proposal(chains, machine_note=note), encoding="utf-8")

    report: dict[str, list[dict[str, Any]]] = {}
    for tier in TIER_CLASSES:
        rows = []
        for entry in chains[tier]:
            available, reason = check_instrument_available(entry.profile, registry)
            rows.append(
                {
                    "profile": entry.profile,
                    "model": entry.model,
                    "effort": entry.effort,
                    "family": entry.family,
                    "access": entry.access,
                    "availability": available,
                    "reason": reason,
                }
            )
        report[tier] = rows
    if json_output:
        output_json(report)
        return
    for tier_name, rows in report.items():
        console.print(f"{tier_name}:")
        if not rows:
            console.print("  (no non-provisional, non-metered entry reaches this tier here)")
        for row in rows:
            effort = f" effort={row['effort']}" if row["effort"] else ""
            mark = "ok " if row["availability"] else "n/a"
            console.print(
                f"  [{mark}] {row['profile']} {row['model']}{effort}  "
                f"({row['family']}, {row['access']})",
                markup=False,
                highlight=False,
                soft_wrap=True,
            )
    if write_proposal is not None:
        console.print(f"Proposal written to {write_proposal} (not installed; merge by hand)")

    @instruments_app.command(name="resolve")
def resolve_score(
    score: Path = typer.Argument(..., help="Score YAML to resolve"),
    json_output: bool = typer.Option(False, "--json", help="Output as JSON"),
) -> None:
    """Show, per sheet, the profile and model each instrument binding resolves to.

    Reads the score the way the conductor does at admission — score aliases,
    capability classes from this machine's class map, per-sheet and
    instrument_map pins — and prints one row per sheet: the alias (or class)
    the sheet was bound through, the profile it resolved to, the model, and
    the fallback chain. This is what a score's diversity claim actually
    configures; the prose says what the author meant.
    """
    from marianne.core.config.job import JobConfig
    from marianne.core.sheet import build_sheets
    from marianne.instruments.classes import load_class_map, resolve_job_classes

    try:
        config = JobConfig.from_yaml(score)
    except Exception as exc:  # noqa: BLE001 — the score's own error is the answer
        output_error(f"cannot load {score}: {exc}")
        raise typer.Exit(1) from exc
    profile_names = set(_load_all_profiles())
    try:
        snapshot = resolve_job_classes(
            config, profile_names, load_class_map(profile_names=profile_names)
        )
    except ValueError as exc:
        output_error(str(exc))
        raise typer.Exit(1) from exc
    sheets = build_sheets(config, classes=snapshot)
    rows: list[ResolvedRow] = []
    for sheet in sheets:
        resolution = sheet.instrument_resolution
        bound_through = sheet.instrument_alias or (
            resolution.requested if resolution is not None else ""
        )
        model = sheet.instrument_config.get("model")
        rows.append({
            "sheet": sheet.num,
            "movement": sheet.movement,
            "voice": sheet.voice,
            "bound_through": bound_through,
            "profile": sheet.instrument_name,
            "model": str(model) if model is not None else None,
            "fallbacks": list(sheet.instrument_fallbacks),
        })
    if json_output:
        output_json({"score": str(score), "sheets": rows})
        return
    table = Table(title=f"Resolved bindings — {config.name}", show_header=True,
                  header_style="bold", padding=(0, 2))
    for col in ("SHEET", "MOVEMENT", "VOICE", "BOUND THROUGH", "PROFILE", "MODEL", "FALLBACKS"):
        table.add_column(col, no_wrap=col != "FALLBACKS")
    for r in rows:
        table.add_row(
            str(r["sheet"]), str(r["movement"]),
            "" if r["voice"] is None else str(r["voice"]),
            r["bound_through"] or "[dim](direct)[/dim]", r["profile"],
            r["model"] or "[dim](profile default)[/dim]",
            ", ".join(r["fallbacks"]) or "[dim]none[/dim]",
        )
    console.print(table)
    distinct = {(r["profile"], r["model"]) for r in rows}
    console.print(f"{len(rows)} sheets, {len(distinct)} distinct profile+model bindings")

    def live_failure(report: LiveReport) -> tuple[str, str] | None:
    """(mode, reason) of the first failed call, or None when the probe passed."""
    for mode in ("plain", "schema"):
        call = report["plain"] if mode == "plain" else report["schema"]
        if call is None or call.get("ok"):
            continue
        reason = (
            call.get("classification")

        )
        return mode, str(reason)

    def live_probe(
    profile: InstrumentProfile,
    *,
    model: str | None = None,
    timeout: float = 180.0,
    record: Path | None = None,
) -> LiveReport:
    """Exercise ``profile`` against the installed instrument, end to end.

    The probe builds the backend through the conductor's own factory and
    classifies each result with the baton's own classifier, so what it
    reports is what a sheet would see: success, the error bucket the chain
    would act on, the wall time, and (in schema mode) whether the validated
    object came back. A profile can pass schema validation and still fail
    every dispatch: a flag the installed CLI rejects, an output shape the
    parser does not read, a provider that is not configured. This is the
    check that catches those.
    """
    import asyncio
    import tempfile
    import time

    from marianne.daemon.baton.backend_pool import _create_backend_for_profile
    from marianne.daemon.baton.musician import _classify_error

    def _one(
        prompt: str, response_format: dict[str, object] | None, mode: str,
    ) -> LiveCall:
        with tempfile.TemporaryDirectory(prefix="mzt-live-probe-") as wd:
            started = time.monotonic()
            raw: dict[str, list[bytes]] = {"stdout": [], "stderr": []}
            try:
                backend = _create_backend_for_profile(
                    profile, working_directory=Path(wd), model=model,
                )
                if response_format is not None:
                    backend.set_response_format(response_format)  # type: ignore[attr-defined]
                if record is not None and hasattr(backend, "set_output_callback"):
                    backend.set_output_callback(
                        lambda stream, chunk: raw.setdefault(stream, []).append(chunk)
                    )
                result = asyncio.run(backend.execute(prompt, timeout_seconds=timeout))
                if record is not None:
                    _write_fixture(
                        record, profile, mode, prompt, response_format, raw,
                        result.exit_code,
                    )
            except Exception as exc:  # noqa: BLE001 — the probe reports any failure as data
                return {
                    "ok": False,
                    "seconds": round(time.monotonic() - started, 2),
                    "classification": "PROBE_ERROR",
                    "error": f"{type(exc).__name__}: {exc}"[:300],
                }
            cls = _classify_error(result)
            row: LiveCall = {
                "ok": bool(result.success),
                "seconds": round(time.monotonic() - started, 2),
                "exit_code": result.exit_code,
                "error_type": result.error_type,
                "classification": (
                    "RATE_LIMITED" if result.rate_limited and not result.success
                    else cls.classification
                ),
                "error_code": cls.error_code,
                "error": (result.error_message or "")[:300] or None,
                "output_head": (result.stdout or "")[:120],
            }
            return row

    plain = _one("Reply with exactly the word: ok", None, "plain")
    ok = bool(plain["ok"])
    schema: LiveCall | None = None
    schema_capable = (
        profile.kind == "http"
        or (
            profile.cli is not None
            and profile.cli.command.json_schema_flag is not None
        )
    )
    if schema_capable:
        schema = _one(_LIVE_PROMPT, _LIVE_SCHEMA, "schema")
        if schema.get("ok"):
            schema["structured_ok"] = _live_structured_ok(str(schema.get("output_head", "")))
            schema["ok"] = bool(schema["structured_ok"])
        ok = ok and bool(schema.get("ok"))
    return {
        "name": profile.name,
        "model": model or profile.default_model,
        "availability": probe_profile(profile).state,
        "plain": plain,
        "schema": schema,
        "ok": ok,
    }

def prices_command(
    json_output: bool = typer.Option(False, "--json", help="Output as JSON"),
    write: bool = typer.Option(
        False,
        "--write",
        help=(
            "Rewrite the drifted price fields on the BUILTIN profiles (the files under "
            "the installed package). User (~/.marianne/instruments/) and venue "
            "(.marianne/instruments/) profiles are reported and left alone: they "
            "are the composer's. Commit the result like any other builtin change."
        ),
    ),
    offline: bool = typer.Option(
        False,
        "--offline",
        help="Do not fetch; compare against the last saved snapshot (see --snapshot).",
    ),
    snapshot: Path | None = typer.Option(
        None,
        "--snapshot",
        help=(
            "Path of a models.dev api.json to compare against instead of fetching "
            "(default cache: ~/.marianne/cache/models.dev.api.json, written on every fetch)."
        ),
    ),
    show_all: bool = typer.Option(
        False,
        "--all",
        help="Also list rows that match, are absent from models.dev, or are unmapped.",
    ),
) -> None:
    """Compare every profile's model prices with models.dev; --write updates the builtins.

    Marianne prices a sheet from the profile's `models` rows; they drift as
    providers reprice. models.dev (MIT, bot-synced) carries input, output,
    cache-read and cache-write prices per model. Exit 0 when nothing drifts,
    1 when a row drifts and was not written, 2 when the snapshot cannot be had.
    """
    from marianne.instruments.loader import profile_source_dirs
    from marianne.instruments.prices import (
        MODELS_DEV_URL,
        PRICE_FIELDS,
        PriceRow,
        apply_drift,
        compare_profiles,
        default_cache_path,
        fetch_snapshot,
        load_layers,
        load_snapshot_file,
        save_snapshot,
        summarize,
    )

    cache = snapshot if snapshot is not None else default_cache_path()
    source: str
    try:
        if snapshot is not None or offline:
            data = load_snapshot_file(cache)
            source = str(cache)
        else:
            data = fetch_snapshot()
            save_snapshot(data, cache)
            source = MODELS_DEV_URL
    except Exception as exc:  # noqa: BLE001 - every failure here is "no snapshot"
        hint = "" if offline or snapshot is not None else f" (--offline reads {cache})"
        output_error(f"cannot load models.dev prices: {exc}{hint}")
        raise typer.Exit(2) from None

    builtins_dir, org_dir, venue_dir = profile_source_dirs()
    builtins, overlays = load_layers(builtins_dir, [org_dir, venue_dir])
    rows = compare_profiles(builtins, data, builtins_dir=builtins_dir)
    rows += compare_profiles(overlays, data, builtins_dir=builtins_dir)
    counts = summarize(rows)

    written: list[PriceRow] = []
    refused: list[PriceRow] = []
    if write:
        written, refused = apply_drift(rows, builtins_dir=builtins_dir)

    if json_output:
        output_json(
            {
                "source": source,
                "snapshot": str(cache),
                "counts": counts,
                "rows": [r.as_dict() for r in rows if show_all or r.status == "drift"],
                "written": [f"{r.profile}:{r.model}" for r in written],
                "refused": [f"{r.profile}:{r.model}" for r in refused],
            }
        )
    else:
        console.print(f"[dim]Prices from {source}[/dim]")
        shown = [r for r in rows if show_all or r.status == "drift"]
        if shown:
            table = Table(title="Model prices vs models.dev (USD per 1K tokens)")
            table.add_column("PROFILE")
            table.add_column("MODEL")
            table.add_column("STATUS")
            table.add_column("FIELD")
            table.add_column("OURS", justify="right")
            table.add_column("MODELS.DEV", justify="right")
            table.add_column("UPDATED")
            for r in shown:
                if r.status != "drift":
                    table.add_row(r.profile, r.model, r.status, "", "", "", r.last_updated or "")
                    continue
                first = True
                for f in PRICE_FIELDS:
                    if f not in r.drifted:
                        continue
                    ours = r.ours.get(f)
                    theirs = r.theirs.get(f)
                    layer = "" if r.builtin else " (user layer)"
                    table.add_row(
                        r.profile if first else "",
                        r.model if first else "",
                        ("drift" + layer) if first else "",
                        f.removeprefix("cost_per_1k_"),
                        "unset" if ours is None else f"{ours:g}",
                        "" if theirs is None else f"{theirs:g}",
                        (r.last_updated or "") if first else "",
                    )
                    first = False
            console.print(table)
        console.print(
            f"{counts['match']} match, {counts['drift']} drift, {counts['absent']} absent from "
            f"models.dev, {counts['unmapped']} unmapped (subscription, local or harness-own ids)"
        )
        if write:
            console.print(f"[green]wrote {len(written)} builtin row(s)[/green]")
            for r in refused:
                console.print(
                    f"[yellow]left alone (not a builtin): {r.profile}:{r.model} "
                    f"in {r.source_path}[/yellow]"
                )
        elif counts["drift"]:
            console.print(
                "[dim]Run with --write to update the builtin rows; "
                "user-layer rows are yours to edit.[/dim]"
            )

    if counts["drift"] and len(written) < counts["drift"]:
        raise typer.Exit(1)
